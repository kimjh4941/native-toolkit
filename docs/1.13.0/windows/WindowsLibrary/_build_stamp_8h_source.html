3// Build-setting guard for consumers of the static library.
4//
5// MSVC records the C runtime choice and the iterator debug level in every
6// object and fails the link (LNK2038) when they disagree. It records nothing
7// about the version of these headers or about the C++/WinRT configuration, so
8// a consumer can link headers of one version against a library of another and
9// only find out at run time. #pragma detect_mismatch adds those records: the
10// linker compares the value the consumer compiled with against the value the
11// library was compiled with, and refuses the link when they differ.
12//
13// Every public and internal header that a consumer can include pulls this in.
14
15// The release these headers and the static library come from. The build script
16// refuses to stamp a distributable with any other version, the way NTK_VERSION_*
17// works for the C ABI. It is a label, not a link-time guard: what the linker
18// compares is NATIVETOOLKIT_ABI_VERSION below.
19#define NATIVETOOLKIT_VERSION_MAJOR 2
20#define NATIVETOOLKIT_VERSION_MINOR 0
21#define NATIVETOOLKIT_VERSION_PATCH 0
22
23#define NATIVETOOLKIT_ABI_VERSION "1"
24
25#pragma detect_mismatch("nativetoolkit_abi_version", NATIVETOOLKIT_ABI_VERSION)
26
27// C++/WinRT's macro configuration has to match across every file that forms one
28// binary, including static libraries. Record the settings that change the
29// generated code.
30#if defined(WINRT_NO_MODULE_LOCK)
31#pragma detect_mismatch("nativetoolkit_winrt_module_lock", "none")
32#elif defined(WINRT_CUSTOM_MODULE_LOCK)
33#pragma detect_mismatch("nativetoolkit_winrt_module_lock", "custom")
34#else
35#pragma detect_mismatch("nativetoolkit_winrt_module_lock", "default")
